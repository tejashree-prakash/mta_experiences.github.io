Subway Line Analysis


The Subway Line Analysis explores how rider experiences, delay patterns, and service perceptions differ across New York City’s subway lines. Our goal is to identify which lines offer smoother commutes, which struggle with chronic delays, and how these differences shape overall rider satisfaction.

Across multiple visualizations, we examine how frequently riders on each line report delays, how often those delays exceed 20 minutes, and how complaint patterns vary by route. The composite score table summarizes overall performance across four key service indicators, offering a system-wide ranking of line families. Together, these interactive graphics reveal both consistent system-wide trends, such as unusually high everyday delays on lines like the 7 or the severe long-delay burden on the Z train.

dist_delay <- scored_data |>
  filter(!is.na(frequency_of_delays)) |>
  count(subway_line, group_color, frequency_of_delays, name = "n") |>
  group_by(subway_line, group_color) |>
  mutate(prop = n / sum(n)) |>
  ungroup() |>
  mutate(
    subway_line = factor(subway_line, levels = unique(subway_line))
  )

dist_delay_plot =
  plot_ly(
  data = dist_delay,
  x = ~subway_line,
  y = ~prop,
  type = "bar",
  color = ~frequency_of_delays,
  colors = viridisLite::viridis(5),
  customdata = ~group_color,
  hovertemplate = paste(
    "Subway Line: %{x}<br>",
    "Line Group: %{customdata}<br>",
    "Proportion of Riders: %{y:.1%}<br>",
    "<extra></extra>"
  )
) |>
   layout(
    barmode = "stack",
    title = list(
      text = "Distribution of Delay Frequency Responses by Subway Line",
      x = 0.5
    ),
    xaxis = list(
      title = "Subway Line",
      tickangle = -45,
      tickmode = "array",
      tickvals = unique(dist_delay$subway_line),
      ticktext = unique(dist_delay$subway_line)
    ),
    yaxis = list(title = "Proportion", tickformat = ".0%"),
    showlegend = TRUE
  )

keep_delays <- c(
  "Rarely",
  "A few times a week",
  "Everyday"
)

heat_data <- scored_data |>
  filter(!is.na(frequency_of_delays)) |>
  filter(frequency_of_delays %in% keep_delays) |>
  count(subway_line, group_color, frequency_of_delays, name = "n") |>
  complete(
    subway_line = unique(subway_directory$subway_line),
    frequency_of_delays = keep_delays,
    fill = list(n = 0)
  ) |>
  group_by(subway_line) |>
  mutate(prop = n / sum(n)) |>
  ungroup() |>
  left_join(subway_directory, by = "subway_line") |>
  mutate(
    subway_line = factor(subway_line, levels = unique(subway_line)),
    frequency_of_delays = factor(frequency_of_delays, levels = keep_delays)
  )

heat_data_plot =

Two different visualizations of delay frequency distribution by subway line: The Stacked bar chart reveals the phenomenon that riders on most lines report delays a few times a week, though the share of everyday delays varies widely. # Proportion of Riders with Long Delays (≥20 min)
# Interactive plotly bar chart

long_categories <- c(
  "20 - 40 min", "40 - 60 min", "60 - 90 min",
  "90 min - 2 hours", "> 60 min"
)

line_order <- subway_directory |>
  arrange(group_color, subway_line) |>
  pull(subway_line)

long_delay_by_line <- scored_data |>
  filter(!is.na(average_length_subway_ride)) |>
  mutate(long_delay = average_length_subway_ride %in% long_categories) |>
  count(subway_line, group_color, hex_color, long_delay, name = "n") |>
  group_by(subway_line, group_color, hex_color) |>
  mutate(prop = n / sum(n)) |>
  filter(long_delay) |>
  ungroup() |>
  mutate(
    subway_line = factor(subway_line, levels = line_order)
  )

plot_ly(
  data = long_delay_by_line,
  x = ~subway_line,
  y = ~prop,
  type = "bar",
  marker = list(color = ~hex_color),
  customdata = ~group_color,
  hovertemplate = paste(
    "<b>Subway Line:</b> %{x}<br>",
    "<b>Long Delay Proportion:</b> %{y:.2f}<br>",
    "<b>Line Group:</b> %{customdata}<br>",
    "<extra></extra>"
  )
) |>
  layout(
    title = "Proportion of Riders Reporting Long Delays (≥20 min)",
    xaxis = list(title = "Subway Line", categoryarray = line_order),
    yaxis = list(title = "Proportion", tickformat = ".0%"),
    bargap = 0.20
  )

This bar chart shows the share of riders on each line who report experiencing long delays of 20 minutes or longer. While most lines cluster between 10-20%, the Z line stands out with the highest proportion, indicating substantially more severe delays among its riders. Interactive hover labels display each line’s exact proportion and color group for more detailed comparison.

What are the subway complaints reported? 

complaint_summary <- complaint_data_time |> 
  count(subway_line_used_most_often, time_of_day, complaint) |> 
  group_by(subway_line_used_most_often, time_of_day) |> 
  mutate(percent = n / sum(n) * 100) |> 
  ungroup()


library(viridis)

all_complaints <- sort(unique(complaint_summary$complaint))
viridis_colors <- viridis(length(all_complaints))
names(viridis_colors) <- all_complaints


times <- unique(complaint_summary$time_of_day)

plot <- plot_ly()
i <- 0

for (t in times) {
  df_t <- complaint_summary |> filter(time_of_day == t)
  
  for (comp in unique(df_t$complaint)) {
    df_tc <- df_t |> filter(complaint == comp)
    
    plot <- plot |> add_trace(
      data = df_tc,
      x = ~percent,
      y = ~subway_line_used_most_often,
      type = "bar",
      orientation = "h",
      name = comp,
      hovertemplate = paste(
        "Time: ", t,
        "<br>Line: %{y}",
        "<br>Complaint: ", comp,
        "<br>Count: %{customdata}",
        "<br>Percent: %{x:.1f}%",
        "<extra></extra>"
      ),
      customdata = ~n,
      marker = list(color = viridis_colors[comp]),
      visible = ifelse(t == times[1], TRUE, FALSE),
      legendgroup = comp
    )
    
    i <- i + 1
  }
}

# Build dropdown buttons
buttons <- list()
for (k in seq_along(times)) {
  vis <- rep(FALSE, length(unique(complaint_summary$complaint)) * length(times))
  
  start <- (k - 1) * length(unique(complaint_summary$complaint)) + 1
  end <- start + length(unique(complaint_summary$complaint)) - 1
  
  vis[start:end] <- TRUE
  
  buttons[[k]] <- list(
    method = "restyle",
    args = list("visible", vis),
    label = times[k]
  )
}

# Final layout with dropdown moved
plot |> layout(
  title = "Complaint Percent by Subway Line and Time of Day",
  xaxis = list(title = "Percent of Complaints"),
  yaxis = list(title = "Subway Line"),
  barmode = "stack",
  # Move legend if needed (optional)
  legend = list(
    x = 1.1,
    y = 0.5
  ),
  updatemenus = list(
    list(
      y = 0.98,        # Move to top right, below title
      x = 0.98,        # Right side
      buttons = buttons,
      type = "dropdown",
      xanchor = "left",  # Anchor to left side
      yanchor = "top"     # Anchor to top
    )
  )
)

This chart provides a detailed breakdown of subway complaints by both the specific transit line and the time of day (morning, afternoon, evening, night).The stacked bar graphs reveal how the distribution of primary complaints (such as delays, overcrowding, and crime) shifts significantly throughout the day, highlighting the distinct challenges during peak versus off-peak hours.The visualization allows for clear comparison across the system, showing which issues are most pressing for each line at different times.
